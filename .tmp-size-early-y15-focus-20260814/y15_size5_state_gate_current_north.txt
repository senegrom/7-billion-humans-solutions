-- Y15 canonical size 5: gate cube events on empty hands and catch both
-- arrival geometries. Conditions fold left-to-right, so this is
-- ((current cube OR north cube) AND empty) OR south shredder.

a:
step n,s
if c == datacube or
 n == datacube and
 myitem == nothing or
 s == shredder:
	pickup c,n
	giveto s
endif
jump a
